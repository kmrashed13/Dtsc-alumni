Supabase Table & Storage Setup SQL

Copy and run this SQL in your Supabase SQL Editor to create dtsc_alumni_members table and member-photos storage bucket:

create table if not exists public.dtsc_alumni_members (
  id uuid primary key default gen_random_uuid(),
  created_at timestamptz not null default now(),
  photo_url text,
  full_name text not null,
  father_name text not null,
  mother_name text not null,
  date_of_birth text not null,
  blood_group text not null,
  contact_number text not null,
  email text not null,
  same_address boolean not null default false,
  present_address jsonb not null,
  permanent_address jsonb not null,
  selected_exams jsonb not null,
  academic_details jsonb not null,
  current_statuses jsonb not null,
  current_status_details jsonb not null,
  suggestions text
);

alter table public.dtsc_alumni_members enable row level security;

create policy "Allow public insert on dtsc_alumni_members"
  on public.dtsc_alumni_members
  for insert
  to anon, authenticated
  with check (true);

insert into storage.buckets (id, name, public)
values ('member-photos', 'member-photos', true)
on conflict (id) do nothing;

create policy "Allow public photo upload to member-photos"
  on storage.objects
  for insert
  to anon, authenticated
  with check (bucket_id = 'member-photos');
